PRISMA MODELS (PostgreSQL)

model User {
  id            String    @id @default(cuid())
  email         String    @unique
  password      String
  fullName      String
  role          Role      @default(STUDENT)
  phone         String?
  dateOfBirth   DateTime?
  isVerified    Boolean   @default(false)
  resetToken    String?
  resetTokenExpiry DateTime?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  subscription  Subscription?
  enrollments   Enrollment[]
  progress      Progress[]
  quizAttempts  QuizAttempt[]
  payments      Payment[]
}

model Subscription {
  id            String    @id @default(cuid())
  userId        String    @unique
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  plan          SubscriptionPlan @default(FREE)
  status        SubscriptionStatus @default(ACTIVE)
  startDate     DateTime  @default(now())
  endDate       DateTime?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
}

enum SubscriptionPlan {
  FREE
  PREMIUM
  ELITE
}

enum SubscriptionStatus {
  ACTIVE
  EXPIRED
  CANCELLED
}

model Course {
  id            String    @id @default(cuid())
  title         String
  description   String
  subject       SubjectEnum
  level         LevelEnum @default(BEGINNER)
  thumbnail     String?
  isActive      Boolean   @default(true)
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  lessons       Lesson[]
  enrollments   Enrollment[]
}

enum SubjectEnum {
  MATHEMATICS
  SCIENCE
  ENGLISH
  LANGUAGES
  COMPUTER_STUDIES
  PHYSICAL_EDUCATION
}

enum LevelEnum {
  BEGINNER
  INTERMEDIATE
  ADVANCED
}

model Lesson {
  id            String    @id @default(cuid())
  courseId      String
  course        Course    @relation(fields: [courseId], references: [id], onDelete: Cascade)
  title         String
  content       String    @db.Text
  videoUrl      String?
  order         Int
  duration      Int?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  progress      Progress[]
  quizzes       Quiz[]
}

model Enrollment {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  courseId      String
  course        Course    @relation(fields: [courseId], references: [id], onDelete: Cascade)
  progress      Float     @default(0)
  completed     Boolean   @default(false)
  enrolledAt    DateTime  @default(now())
  completedAt   DateTime?
  lastAccessed  DateTime  @default(now())
}

model Progress {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  lessonId      String
  lesson        Lesson    @relation(fields: [lessonId], references: [id], onDelete: Cascade)
  completed     Boolean   @default(false)
  completedAt   DateTime?
  timeSpent     Int?
  lastPosition  Int?
  updatedAt     DateTime  @updatedAt

  @@unique([userId, lessonId])
}

model Quiz {
  id            String    @id @default(cuid())
  lessonId      String
  lesson        Lesson    @relation(fields: [lessonId], references: [id], onDelete: Cascade)
  title         String
  description   String?
  passingScore  Int       @default(70)
  timeLimit     Int?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  questions     Question[]
  attempts      QuizAttempt[]
}

model Question {
  id            String    @id @default(cuid())
  quizId        String
  quiz          Quiz      @relation(fields: [quizId], references: [id], onDelete: Cascade)
  text          String
  type          QuestionType @default(MULTIPLE_CHOICE)
  options       String[]  // JSON array
  correctAnswer String
  explanation   String?
  order         Int
  points        Int       @default(1)
}

enum QuestionType {
  MULTIPLE_CHOICE
  TRUE_FALSE
  FILL_IN_BLANK
  MATCHING
}

model QuizAttempt {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  quizId        String
  quiz          Quiz      @relation(fields: [quizId], references: [id], onDelete: Cascade)
  score         Int
  passed        Boolean
  answers       Json
  startedAt     DateTime  @default(now())
  completedAt   DateTime?
}

model Payment {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  amount        Float
  currency      String    @default("NGN")
  status        PaymentStatus @default(PENDING)
  reference     String    @unique
  description   String?
  paymentMethod String?
  paidAt        DateTime?
  metadata      Json?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
}

enum PaymentStatus {
  PENDING
  COMPLETED
  FAILED
  REFUNDED
}

model ContactMessage {
  id            String    @id @default(cuid())
  name          String
  email         String
  phone         String?
  subject       String
  message       String    @db.Text
  isRead        Boolean   @default(false)
  createdAt     DateTime  @default(now())
}

enum Role {
  STUDENT
  PARENT
  COACH
  ADMIN
}

// src/services/api.ts
const API_BASE = import.meta.env.VITE_API_URL || 'http://localhost:5000/api';

export const api = {
  auth: {
    login: (email: string, password: string) => 
      axios.post(`${API_BASE}/auth/login`, { email, password }),
    register: (data: any) => 
      axios.post(`${API_BASE}/auth/register`, data),
    forgotPassword: (email: string) => 
      axios.post(`${API_BASE}/auth/forgot-password`, { email }),
    resetPassword: (token: string, password: string) => 
      axios.post(`${API_BASE}/auth/reset-password`, { token, password }),
    me: () => 
      axios.get(`${API_BASE}/auth/me`, { headers: { Authorization: `Bearer ${localStorage.getItem('token')}` } }),
  },
  courses: {
    getAll: () => axios.get(`${API_BASE}/courses`),
    getById: (id: string) => axios.get(`${API_BASE}/courses/${id}`),
    getEnrolled: () => axios.get(`${API_BASE}/courses/enrolled`),
    enroll: (courseId: string) => axios.post(`${API_BASE}/courses/enroll`, { courseId }),
  },
  lessons: {
    getByCourse: (courseId: string) => axios.get(`${API_BASE}/lessons/course/${courseId}`),
    getById: (id: string) => axios.get(`${API_BASE}/lessons/${id}`),
    markComplete: (lessonId: string) => axios.post(`${API_BASE}/lessons/complete`, { lessonId }),
  },
  progress: {
    getDashboard: () => axios.get(`${API_BASE}/progress/dashboard`),
    getCourse: (courseId: string) => axios.get(`${API_BASE}/progress/course/${courseId}`),
  },
  quizzes: {
    getByLesson: (lessonId: string) => axios.get(`${API_BASE}/quizzes/lesson/${lessonId}`),
    getById: (id: string) => axios.get(`${API_BASE}/quizzes/${id}`),
    submit: (quizId: string, answers: any) => axios.post(`${API_BASE}/quizzes/submit`, { quizId, answers }),
    getUpcoming: () => axios.get(`${API_BASE}/quizzes/upcoming`),
  },
  subscription: {
    getCurrent: () => axios.get(`${API_BASE}/subscription/current`),
    upgrade: (plan: string) => axios.post(`${API_BASE}/subscription/upgrade`, { plan }),
    cancel: () => axios.post(`${API_BASE}/subscription/cancel`),
  },
  certificates: {
    getAll: () => axios.get(`${API_BASE}/certificates`),
    getById: (id: string) => axios.get(`${API_BASE}/certificates/${id}`),
  },
  academy: {
    getPrograms: () => axios.get(`${API_BASE}/academy/programs`),
    registerScreening: (data: any) => axios.post(`${API_BASE}/academy/screening/register`, data),
  },
};
